01 · Conteúdo

Do SELECT ao relatório

SELECT + FROM

Escolhem colunas e origem.

SELECT nome, preco FROM produto;

WHERE

Filtra linhas por condição.

WHERE preco > 10

ORDER BY

Ordena o resultado.

ORDER BY preco DESC

LIKE / IN / BETWEEN

Facilitam filtros por padrão, lista ou faixa.

WHERE nome LIKE '%café%'

Agregações

Resumem valores.

COUNT AVG SUM MIN MAX

GROUP BY + HAVING

Agrupam dados e filtram grupos.

GROUP BY categoria HAVING AVG(preco) > 10

JOIN

Combina tabelas relacionadas.

JOIN categoria c ON c.id_categoria = p.id_categoria

CASE

Cria classificações condicionais no resultado.

CASE WHEN estoque < 5 THEN 'REPOR' END

02 · Demonstração

Uma consulta completa

SELECT
  c.nome_categoria,
  COUNT(p.id_produto) AS total_produtos,
  AVG(p.preco) AS preco_medio
FROM produto p
JOIN categoria c
  ON c.id_categoria = p.id_categoria
GROUP BY c.nome_categoria
HAVING AVG(p.preco) > 10
ORDER BY preco_medio DESC;
